The smallest complete program that uses the retained UI framework. It opens a window, picks a theme preset, builds a vertical box holding a title label, a primary button, a switch and a slider, and runs the two per-frame calls every UI needs: =lrg_canvas_handle_input()= and =lrg_canvas_render()=. Everything else (animation, focus rings, gamepad navigation, shadows) comes for free.

When you have this running, continue with the two full walkthroughs:

- [[file:ui-showcase.org][UI Showcase]]: every widget in an application layout, runtime theme and accent switching, motion presets, gradient cards from style rules, toasts, a modal dialog and a 10,000-row list.
- [[file:ui-game-menu.org][Game UI Menu]]: game menus over a live scene, a YAML theme, a fixed virtual resolution with letterboxing, a modal pause menu and gamepad-first navigation.

** The program
:PROPERTIES:
:CUSTOM_ID: ui-basics-program
:END:
#+begin_src C
/* ui-basics.c - the smallest complete libregnum UI program */

#include <libregnum.h>
#include <graylib.h>
#include <raylib.h>

/* on_play_clicked: a primary button was clicked (mouse, Enter, Space or gamepad A) */
static void
on_play_clicked (LrgButton *button,
                 gpointer   user_data)
{
	LrgLabel *status = LRG_LABEL (user_data);

	lrg_label_set_text (status, "Play clicked!");
}

/* on_bloom_toggled: the switch flipped; the new state comes with the signal */
static void
on_bloom_toggled (LrgCheckbox *checkbox,
                  gboolean     checked,
                  gpointer     user_data)
{
	g_print ("bloom %s\n", checked ? "on" : "off");
}

/* on_volume_changed: the slider moved; read the value back from it */
static void
on_volume_changed (LrgSlider *slider,
                   gpointer   user_data)
{
	g_print ("volume %.0f\n", lrg_slider_get_value (slider));
}

int
main (int   argc,
      char *argv[])
{
	g_autoptr(GrlWindow) window = NULL;
	g_autoptr(LrgTheme)  theme = NULL;
	g_autoptr(LrgCanvas) canvas = NULL;
	LrgVBox             *root;
	LrgLabel            *status;
	LrgButton           *play;
	LrgCheckbox         *bloom;
	LrgSlider           *volume;

	/* 1. A window, then a theme preset as the process-wide default */
	SetConfigFlags (FLAG_WINDOW_RESIZABLE | FLAG_MSAA_4X_HINT);
	window = grl_window_new (640, 400, "UI basics");
	grl_window_set_target_fps (window, 60);

	theme = lrg_theme_new_preset (argc > 1 ? argv[1] : "midnight");
	if (theme == NULL)
		theme = lrg_theme_new_preset ("midnight");
	lrg_theme_set_default (theme);

	/* 2. The canvas is the root of the widget tree; let it fill the
	 *    window with the theme's WINDOW colour */
	canvas = lrg_canvas_new ();
	lrg_canvas_set_paint_background (canvas, TRUE);

	/* 3. A vertical box stacks its children with padding and spacing */
	root = lrg_vbox_new ();
	lrg_container_set_padding (LRG_CONTAINER (root), 24.0f);
	lrg_container_set_spacing (LRG_CONTAINER (root), 14.0f);
	lrg_container_add_child (LRG_CONTAINER (canvas), LRG_WIDGET (root));

	status = lrg_label_new ("Hello, libregnum");
	lrg_widget_add_class (LRG_WIDGET (status), "title");
	lrg_container_add_child (LRG_CONTAINER (root), LRG_WIDGET (status));

	play = lrg_button_new ("Play");
	lrg_widget_add_class (LRG_WIDGET (play), "primary");
	lrg_widget_set_halign (LRG_WIDGET (play), LRG_ALIGN_START);
	g_signal_connect (play, "clicked", G_CALLBACK (on_play_clicked), status);
	lrg_container_add_child (LRG_CONTAINER (root), LRG_WIDGET (play));

	bloom = lrg_checkbox_new_with_style ("Enable bloom", LRG_CHECKBOX_STYLE_SWITCH);
	g_signal_connect (bloom, "toggled", G_CALLBACK (on_bloom_toggled), NULL);
	lrg_container_add_child (LRG_CONTAINER (root), LRG_WIDGET (bloom));

	volume = lrg_slider_new_with_range (0.0, 100.0, 1.0);
	lrg_slider_set_value (volume, 65.0);
	g_signal_connect (volume, "value-changed", G_CALLBACK (on_volume_changed), NULL);
	lrg_container_add_child (LRG_CONTAINER (root), LRG_WIDGET (volume));

	/* 4. The frame loop: follow the window size, feed input, render */
	while (!grl_window_should_close (window))
	{
		gfloat w = (gfloat) GetScreenWidth ();
		gfloat h = (gfloat) GetScreenHeight ();

		lrg_widget_set_size (LRG_WIDGET (canvas), w, h);
		lrg_widget_set_size (LRG_WIDGET (root), w, h);

		lrg_canvas_handle_input (canvas);

		grl_window_begin_drawing (window);
		lrg_canvas_render (canvas);
		grl_window_end_drawing (window);
	}

	/* The canvas owns its children; drop our own references */
	g_object_unref (volume);
	g_object_unref (bloom);
	g_object_unref (play);
	g_object_unref (status);
	g_object_unref (root);
	return 0;
}
#+end_src

** Step by step
:PROPERTIES:
:CUSTOM_ID: ui-basics-steps
:END:
*** Theme first
:PROPERTIES:
:CUSTOM_ID: ui-basics-theme
:END:
=lrg_theme_new_preset()= returns one of the built-in themes (=midnight=, =daylight=, =catppuccin-mocha=, =catppuccin-latte=, =high-contrast=; the list is also available from =lrg_theme_list_presets()=), or =NULL= for an unknown name. =lrg_theme_set_default()= installs it process-wide and takes its own reference, so the local =g_autoptr= is fine. Widgets read the default theme every frame: calling =lrg_theme_set_default()= again later restyles the running UI, and =lrg_theme_set_accent()= recolours every accent-coloured part from one colour. See [[file:../guides/theming.org][Theming Guide]].

*** The canvas and the tree
:PROPERTIES:
:CUSTOM_ID: ui-basics-tree
:END:
=LrgCanvas= is the root of a widget tree and the thing that turns input into events and widgets into draw calls. =lrg_canvas_set_paint_background()= makes it fill itself with the theme's window colour; leave it off (the default) when the UI is drawn over a game scene.

=LrgVBox= stacks its children top to bottom with the container's padding and spacing. Children stretch to the box's width unless their horizontal alignment says otherwise, which is why the button gets =LRG_ALIGN_START=. =lrg_container_add_child()= takes its own reference to the child.

*** Widgets and signals
:PROPERTIES:
:CUSTOM_ID: ui-basics-widgets
:END:
| Widget                                                             | Signal          | Handler signature                                    |
|--------------------------------------------------------------------+-----------------+------------------------------------------------------|
| =lrg_button_new ("Play")= + class =primary=                        | =clicked=       | =(LrgButton *, gpointer)=                            |
| =lrg_checkbox_new_with_style (label, LRG_CHECKBOX_STYLE_SWITCH)=   | =toggled=       | =(LrgCheckbox *, gboolean checked, gpointer)=        |
| =lrg_slider_new_with_range (min, max, step)=                       | =value-changed= | =(LrgSlider *, gpointer)=, read =lrg_slider_get_value()= |

Style classes such as =primary= (accent fill), =flat=, =danger=, =title=, =heading= and =muted= are interpreted by the theme; add them with =lrg_widget_add_class()=. A button fires =clicked= on a mouse click and on Enter, Space or gamepad A while focused; =Tab= and the arrow keys or D-pad move focus. See [[file:../modules/ui/widgets/button.org][Button]], [[file:../modules/ui/widgets/checkbox.org][Checkbox]], [[file:../modules/ui/widgets/slider.org][Slider]], [[file:../modules/ui/widgets/label.org][Label]] and [[file:../modules/ui/widgets/vbox.org][VBox]].

*** The frame loop
:PROPERTIES:
:CUSTOM_ID: ui-basics-loop
:END:
1. Size the canvas and the root box to the window, so resizing works. Setting an unchanged size costs nothing.
2. =lrg_canvas_handle_input()=: polls mouse, keyboard and gamepad, dispatches events (your signal handlers run here) and advances every spring and appear animation by the frame time. Call it every frame.
3. =lrg_canvas_render()= between =grl_window_begin_drawing()= and =grl_window_end_drawing()=: lays out what changed, records the tree into a paint list and draws it.

To draw the UI over a game, render your scene first, then call =lrg_canvas_render()= in the same drawing block. To draw at a fixed virtual resolution, render into a render texture and install a pointer transform; [[file:ui-game-menu.org#game-menu-virtual][Game UI Menu]] shows how.
