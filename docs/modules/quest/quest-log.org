#+begin_src C
LrgQuestInstance * lrg_quest_log_start_quest (LrgQuestLog *self,
                                              LrgQuestDef *quest_def);
#+end_src

Starts a new quest from a definition. Creates an instance and adds it to the active quests. This legacy entry point performs no level, prerequisite, exclusivity or repeat checks and returns =NULL= for any quest that is active or has a completion record. When its instance completes the last stage it moves to the completed set automatically and a completion record with an unknown time (=last= = 0) is created. Server-authoritative games should use =lrg_quest_log_start_quest_checked()= instead.

*Parameters:*

- =self= - An =LrgQuestLog=
- =quest_def= - Quest definition to start

*Returns:* The quest instance (transfer none, nullable) - =NULL= on failure

** Signals
:PROPERTIES:
:CUSTOM_ID: signals
:END:
- =quest-started (LrgQuestInstance *quest)= - a quest was started (not emitted while restoring a snapshot)
- =quest-completed (LrgQuestInstance *quest)= - a legacy quest auto-completed, or a checked quest was turned in
- =quest-ready (LrgQuestInstance *quest)= - a checked quest finished its last stage and waits for =lrg_quest_log_turn_in()=
- =quest-failed=, =quest-abandoned=, =objective-updated=, =tracked-changed= - unchanged

An instance that is no longer the active instance for its ID (abandoned, turned in, or replaced by a restart) never drives the log again.

** Availability and Turn-in
:PROPERTIES:
:CUSTOM_ID: availability-and-turn-in
:END:
All errors below are in the =LRG_PROGRESSION_ERROR= domain and leave the log unchanged.

*** lrg_quest_log_can_start()
:PROPERTIES:
:CUSTOM_ID: lrg_quest_log_can_start
:END:
#+begin_src C
gboolean lrg_quest_log_can_start (LrgQuestLog      *self,
                                  LrgQuestDef      *quest_def,
                                  guint             level,
                                  LrgResetSchedule *schedule,
                                  gint64            now,
                                  GError          **error);
#+end_src

Full availability check. Rules are evaluated in this order and the first failure is reported:

1. Already active - =LRG_PROGRESSION_ERROR_DUPLICATE=
2. =level= below =min-level= - =LRG_PROGRESSION_ERROR_REQUIREMENT=
3. A prerequisite has no completion record - =LRG_PROGRESSION_ERROR_REQUIREMENT= (message "... requires <id>")
4. An exclusive quest is active or completed - =LRG_PROGRESSION_ERROR_REQUIREMENT=
5. Completed and =repeat= is =NONE= - =LRG_PROGRESSION_ERROR_DUPLICATE=
6. Completed and repeatable with =schedule= =NULL= - =LRG_PROGRESSION_ERROR_INVALID=; completed and the reset period (daily or weekly per =repeat=) of the last completion is not earlier than the period of =now= - =LRG_PROGRESSION_ERROR_NOT_READY=
7. =LRG_QUEST_LOG_MAX_ACTIVE= (256) quests already active, or a new completion record would exceed =LRG_QUEST_LOG_MAX_COMPLETED= (8192) - =LRG_PROGRESSION_ERROR_LIMIT=

Rule 7 is an engine addition that guarantees every reachable log can be persisted. The =check_prerequisites= virtual method is not called. A completion recorded at an unknown time (0) counts as the period containing the Unix epoch.

*Parameters:*

- =self= - An =LrgQuestLog=
- =quest_def= - Quest definition
- =level= - Character level
- =schedule= - (nullable) Reset schedule; required only for repeatable quests that were completed before
- =now= - Trusted server Unix seconds
- =error= - (nullable) Error return

*Returns:* =TRUE= if the quest may be started

*** lrg_quest_log_start_quest_checked()
:PROPERTIES:
:CUSTOM_ID: lrg_quest_log_start_quest_checked
:END:
#+begin_src C
LrgQuestInstance *lrg_quest_log_start_quest_checked (LrgQuestLog      *self,
                                                     LrgQuestDef      *quest_def,
                                                     guint             level,
                                                     LrgResetSchedule *schedule,
                                                     gint64            now,
                                                     GError          **error);
#+end_src

Runs =lrg_quest_log_can_start()= and starts the quest in turn-in mode: when its last stage completes the instance stays active in state =LRG_QUEST_STATE_COMPLETE= and =quest-ready= is emitted. A restarted repeatable quest keeps its completion record as history, so it is both active and completed until turned in again.

*Parameters:*

- Same as =lrg_quest_log_can_start()=

*Returns:* The new active instance (transfer none), or =NULL= with =error= set

*Errors:* Same as =lrg_quest_log_can_start()=

*** lrg_quest_log_turn_in()
:PROPERTIES:
:CUSTOM_ID: lrg_quest_log_turn_in
:END:
#+begin_src C
gboolean lrg_quest_log_turn_in (LrgQuestLog *self,
                                const gchar *quest_id,
                                gint64       now,
                                GError     **error);
#+end_src

Turns in an active quest whose instance is complete: the instance moves to the completed set (replacing an older completed instance), the completion count is incremented (saturating at =G_MAXUINT=), =now= becomes the last completion time, a tracked quest is untracked and =quest-completed= is emitted. Rewards are not granted; call =lrg_quest_def_grant_rewards()= after success.

*Parameters:*

- =self= - An =LrgQuestLog=
- =quest_id= - Quest ID
- =now= - Trusted server Unix seconds
- =error= - (nullable) Error return

*Returns:* =TRUE= on success

*Errors:* =LRG_PROGRESSION_ERROR_NOT_FOUND= if the quest is not active; =LRG_PROGRESSION_ERROR_REQUIREMENT= if its instance is not in state =COMPLETE= (including failed quests)

** Persistence
:PROPERTIES:
:CUSTOM_ID: persistence
:END:
Snapshot type (=LRG_QUEST_LOG_VARIANT_TYPE=): =(a(suau)a(sux))=

| Part | Element | Meaning |
|------+---------+---------|
| active | =(s u au)= | quest ID, current stage, progress of every objective of that stage |
| completed | =(s u x)= | quest ID, completion count, last completion Unix seconds |

Both arrays are sorted by quest ID. The stage equals the stage count when every stage is done and the quest waits for turn-in; its progress array is then empty. Progress values are clamped to their targets. The tracked quest and instance states (for example =FAILED=) are not persisted. Limits: 256 active entries, 8192 completed entries, 16 progress values per entry, IDs of 1 to 128 bytes of UTF-8.

*** lrg_quest_log_new_from_variant()
:PROPERTIES:
:CUSTOM_ID: lrg_quest_log_new_from_variant
:END:
#+begin_src C
LrgQuestLog *lrg_quest_log_new_from_variant (GVariant    *variant,
                                             GHashTable  *defs,
                                             GError     **error);
#+end_src

Restores a log. Every entry is validated before anything is built, so a bad entry rejects the whole snapshot with no side effects. Active quests are restored in turn-in mode without emitting =quest-started=; an entry at stage == stage count is restored in state =COMPLETE=. Missing trailing progress values are treated as 0. Completed quests get an unconnected completed instance so =lrg_quest_log_get_quest()= keeps working.

*Parameters:*

- =variant= - Snapshot
- =defs= - (element-type utf8 LrgQuestDef) Definitions by ID
- =error= - (nullable) Error return

*Returns:* The restored log (transfer full, nullable)

*Errors:* =LRG_PROGRESSION_ERROR_NOT_FOUND= for a quest ID missing from =defs= (or mapped to a non-=LrgQuestDef=); =LRG_PROGRESSION_ERROR_INVALID= for a wrong type string, non-normal form (including invalid UTF-8), too many entries, an empty/overlong ID, a duplicate ID within either array, a stage beyond the stage count, more than 16 progress values or more values than the stage has objectives, a value above its objective's target, or a completion count of 0

** Server Example
:PROPERTIES:
:CUSTOM_ID: server-example

#+begin_src C
g_autoptr(GError) error = NULL;
LrgQuestInstance *daily;

daily = lrg_quest_log_start_quest_checked (log, bounty_def, level, schedule, now, &error);
if (daily == NULL)
{
    if (g_error_matches (error, LRG_PROGRESSION_ERROR, LRG_PROGRESSION_ERROR_NOT_READY))
        notify_client (client, "Come back after the daily reset (%" G_GINT64_FORMAT " s)",
                       lrg_reset_schedule_get_seconds_until_reset (schedule, LRG_RESET_PERIOD_DAILY, now));
    return;
}

/* Later, when the player talks to the quest giver. */
if (!lrg_quest_log_turn_in (log, "wolf_bounty", now, &error))
    return notify_client (client, "%s", error->message);
lrg_quest_def_grant_rewards (bounty_def, character);

/* Save. */
g_autoptr(GVariant) snapshot = lrg_quest_log_to_variant (log);
store_character_record (store, character_id, "quests", snapshot);

** Example
:PROPERTIES:
:CUSTOM_ID: example
:END:
#+begin_src C
/* Create quest log */
g_autoptr(LrgQuestLog) journal = lrg_quest_log_new();

/* NPC offers a quest */
g_autoptr(LrgQuestDef) dragon_quest = load_quest_def("slay_dragon");
LrgQuestInstance *instance = lrg_quest_log_start_quest(journal, dragon_quest);
if (instance) {
    g_print("Quest accepted: %s\n", lrg_quest_def_get_name(dragon_quest));
}

/* Player tracks this quest for HUD */
lrg_quest_log_track_quest(journal, "slay_dragon");

/* Later, display tracked quest on HUD */
LrgQuestInstance *tracked = lrg_quest_log_get_tracked_quest(journal);
if (tracked) {
    LrgQuestObjective *obj = lrg_quest_instance_get_current_objective(tracked);
    if (obj) {
        g_print("Objective: %s\n", lrg_quest_objective_get_description(obj));
    }
}

/* Check if quest is active */
if (lrg_quest_log_is_quest_active(journal, "collect_herbs")) {
    g_print("Herb gathering is still in progress\n");
}

/* Player completes a quest */
LrgQuestInstance *completed_quest = lrg_quest_log_get_quest(journal, "collect_herbs");
if (completed_quest) {
    lrg_quest_instance_complete(completed_quest);

    if (lrg_quest_log_is_quest_completed(journal, "collect_herbs")) {
        g_print("Herb quest finished!\n");
    }
}

/* View all active quests */
GList *active = lrg_quest_log_get_active_quests(journal);
g_print("Active quests (%d):\n", lrg_quest_log_get_active_count(journal));
for (GList *l = active; l != NULL; l = l->next) {
    LrgQuestInstance *q = l->data;
    LrgQuestDef *def = lrg_quest_instance_get_quest_def(q);
    g_print("  - %s\n", lrg_quest_def_get_name(def));
}
g_list_free(active);

/* View all completed quests */
GList *completed = lrg_quest_log_get_completed_quests(journal);
g_print("Completed quests (%d):\n", lrg_quest_log_get_completed_count(journal));
for (GList *l = completed; l != NULL; l = l->next) {
    LrgQuestInstance *q = l->data;
    LrgQuestDef *def = lrg_quest_instance_get_quest_def(q);
    g_print("  - %s\n", lrg_quest_def_get_name(def));
}
g_list_free(completed);

/* Player gives up */
lrg_quest_log_abandon_quest(journal, "boring_fetch_quest");
#+end_src
