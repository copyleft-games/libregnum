** Overview
:PROPERTIES:
:CUSTOM_ID: overview
:END:
LrgProceduralAudio wraps GrlAudioStream to provide real-time audio synthesis. Subclass it and override the =generate()= virtual method to create custom synthesizers, sound effects generators, or audio analyzers.

#+begin_src C
/* Create custom synthesizer by subclassing */
struct _MySynth {
    LrgProceduralAudio parent_instance;
    gfloat frequency;
    gfloat phase;
};

G_DEFINE_TYPE (MySynth, my_synth, LRG_TYPE_PROCEDURAL_AUDIO)

static void
my_synth_generate (LrgProceduralAudio *audio,
                   gfloat             *buffer,
                   gint                frame_count)
{
    MySynth *self = MY_SYNTH (audio);
    guint sample_rate = lrg_procedural_audio_get_sample_rate (audio);

    for (gint i = 0; i < frame_count; i++) {
        buffer[i] = sinf (self->phase);
        self->phase += 2.0f * G_PI * self->frequency / sample_rate;
    }
}
#+end_src

*** Update
:PROPERTIES:
:CUSTOM_ID: update-required
:END:
Harmless to call every frame.  On the audio thread (the default, see [[#audio-thread][The audio thread]]) it only runs the destroy notifies of commands the audio thread has run; with =audio-thread= =FALSE= it is required and keeps the buffer filled on the caller's thread:

#+begin_src C
/* In game loop */
lrg_procedural_audio_update (audio);
#+end_src

*** The audio thread
:PROPERTIES:
:CUSTOM_ID: audio-thread
:END:
By default (the construct-only =audio-thread= property, =TRUE=) =generate()= runs on the audio device's thread: graylib's stream callback (=grl_audio_stream_set_callback()=) pulls a few hundred frames at a time while the source plays, just before the device plays them.  Before this change =update()= rendered a whole 4096-frame buffer (about 93 ms of audio) on the frame the device consumed one, about one frame in five.  Now synthesis never costs the frame, and a parameter change is heard within one device period instead of up to two buffers later.

What this asks of a subclass:

- =generate()= must not block, allocate heavily, or call graylib's or raylib's audio functions (it runs inside raylib's mixer, raylib's audio lock held).
- The state =generate()= reads belongs to the audio thread.  Change it with =lrg_procedural_audio_queue_command (audio, func, user_data, destroy)=: =func (audio, user_data)= runs on the audio thread just before the next block, after every command queued before it (a lock-free list, =src/audio/lrg-audio-thread-private.h=); =destroy= runs later on the game thread (from the next call on the source, such as =update()=).  While the source does not play (never played, stopped or paused: raylib then does not call the callback) commands run at once on the caller's thread.  Plain atomics (=g_atomic_int_set=) are fine for single values too.
- The audio thread lets go of the source in =dispose= (before any =finalize=), running the commands still pending; free what =generate()= reads in =finalize=, not in =dispose=.
- =lrg_procedural_audio_get_audio_thread()= tells whether the audio thread renders (the property is on, the device was open when the source was made, and graylib had a callback slot).  Otherwise the old model applies and =update()= pumps 4096-frame buffers on the caller's thread.

#+begin_src C
static void
apply_frequency (LrgProceduralAudio *audio, gpointer user_data)
{
    MY_SYNTH (audio)->frequency = *(gfloat *)user_data;   /* on the audio thread */
}

void
my_synth_set_frequency (MySynth *self, gfloat frequency)
{
    lrg_procedural_audio_queue_command (LRG_PROCEDURAL_AUDIO (self), apply_frequency,
                                        g_memdup2 (&frequency, sizeof frequency), g_free);
}

/* the old model (generate () on the thread that calls update ()) */
g_autoptr(LrgProceduralAudio) pumped = g_object_new (MY_TYPE_SYNTH, "audio-thread", FALSE, NULL);
#+end_src

lrg_audio_manager_play_procedural (manager, "lead-synth");

** Example: Complete Synthesizer
:PROPERTIES:
:CUSTOM_ID: example-complete-synthesizer
:END:
#+begin_src C
#include <libregnum.h>
#include <math.h>

/* ADSR Envelope */
typedef struct {
    gfloat attack;
    gfloat decay;
    gfloat sustain;
    gfloat release;
    gfloat time;
    gboolean released;
} Envelope;

/* Simple Synth */
#define MY_TYPE_SIMPLE_SYNTH (my_simple_synth_get_type ())
G_DECLARE_FINAL_TYPE (MySimpleSynth, my_simple_synth, MY, SIMPLE_SYNTH, LrgProceduralAudio)

struct _MySimpleSynth {
    LrgProceduralAudio parent_instance;
    gfloat frequency;
    gfloat phase;
    Envelope env;
};

static gfloat
envelope_value (Envelope *env, guint sample_rate)
{
    gfloat value = 0.0f;
    gfloat dt = 1.0f / sample_rate;

    if (env->time < env->attack) {
        value = env->time / env->attack;
    } else if (env->time < env->attack + env->decay) {
        gfloat t = (env->time - env->attack) / env->decay;
        value = 1.0f - t * (1.0f - env->sustain);
    } else if (!env->released) {
        value = env->sustain;
    } else {
        gfloat release_time = env->time - env->attack - env->decay;
        value = env->sustain * (1.0f - release_time / env->release);
        if (value < 0.0f) value = 0.0f;
    }

    env->time += dt;
    return value;
}

static void
my_simple_synth_generate (LrgProceduralAudio *audio,
                           gfloat             *buffer,
                           gint                frame_count)
{
    MySimpleSynth *self = MY_SIMPLE_SYNTH (audio);
    guint sample_rate = lrg_procedural_audio_get_sample_rate (audio);
    gfloat phase_inc = 2.0f * G_PI * self->frequency / sample_rate;

    for (gint i = 0; i < frame_count; i++) {
        gfloat osc = sinf (self->phase);
        gfloat env = envelope_value (&self->env, sample_rate);
        buffer[i] = osc * env;

        self->phase += phase_inc;
        if (self->phase >= 2.0f * G_PI)
            self->phase -= 2.0f * G_PI;
    }
}

G_DEFINE_TYPE (MySimpleSynth, my_simple_synth, LRG_TYPE_PROCEDURAL_AUDIO)

static void
my_simple_synth_class_init (MySimpleSynthClass *klass)
{
    LRG_PROCEDURAL_AUDIO_CLASS (klass)->generate = my_simple_synth_generate;
}

static void
my_simple_synth_init (MySimpleSynth *self)
{
    self->frequency = 440.0f;
    self->phase = 0.0f;
    self->env = (Envelope){ 0.01f, 0.1f, 0.7f, 0.3f, 0.0f, FALSE };
}

void
my_simple_synth_note_on (MySimpleSynth *self, gfloat frequency)
{
    self->frequency = frequency;
    self->phase = 0.0f;
    self->env.time = 0.0f;
    self->env.released = FALSE;
}

void
my_simple_synth_note_off (MySimpleSynth *self)
{
    self->env.released = TRUE;
}
#+end_src

#+begin_src C
static void
stereo_generate (LrgProceduralAudio *audio,
                 gfloat             *buffer,
                 gint                frame_count)
{
    for (gint i = 0; i < frame_count; i++) {
        gfloat sample = /* generate sample */;

        /* Interleaved stereo: L, R, L, R, ... */
        buffer[i * 2 + 0] = sample;      /* Left */
        buffer[i * 2 + 1] = sample;      /* Right */
    }
}
