=LrgWidget= (=src/ui/lrg-widget.h=, derivable =GObject=) is the base of every element in the retained-mode UI tree. It owns the widget's rectangle and visibility, its layout hints, its style classes and interaction state, two springs (hover and press) that animate its looks, an optional entrance animation, an optional scoped theme, and the virtual methods subclasses implement: =paint()=, =style_defaults()=, =measure()=, =arrange()= and =handle_event()=. =LrgWidget= itself is instantiable and draws nothing, which makes a plain =g_object_new (LRG_TYPE_WIDGET, NULL)= a handy spacer.

** Virtual methods
:PROPERTIES:
:CUSTOM_ID: widget-vfuncs
:END:
| VFunc | Signature | Default | Purpose |
|-------+-----------+---------+---------|
| =paint= | =void (*paint) (LrgWidget *self, LrgPaintList *list)= | nothing (containers: paint children in order) | Record the widget at its *world* position. Opacity, appear transforms and the focus ring are applied around it by =lrg_widget_paint()=. New widgets implement this. |
| =style_defaults= | =void (*style_defaults) (LrgWidget *self, LrgUIPart part, LrgWidgetState state, LrgPartLook *look)= | the =FOCUS= ring look; other parts invisible | Fill the built-in look of =part= in exactly =state= from the theme. Chain up for parts you don't handle. See [[file:styling.org::#styling-defaults][styling]]. |
| =measure= | =void (*measure) (LrgWidget *self, gfloat *w, gfloat *h)= | the current size | Preferred size *without* margin. See [[file:layout.org::#layout-measure][layout]]. |
| =arrange= | =void (*arrange) (LrgWidget *self)= | nothing (containers: =layout_children()=) | Called after the rectangle was assigned or the widget was marked for layout. |
| =handle_event= | =gboolean (*handle_event) (LrgWidget *self, const LrgUIEvent *event)= | =FALSE= (containers: forward non-bubbling events to children) | Return =TRUE= to consume and stop bubbling. See [[file:input-and-focus.org][input]]. |
| =draw= | =void (*draw) (LrgWidget *self)= | renders =paint()= output immediately | *Legacy* immediate drawing with graylib. Widgets overriding only =draw= are painted through a paint callback. |

The class has five reserved pointers for future vfuncs.

** Looks and painting
:PROPERTIES:
:CUSTOM_ID: widget-looks
:END:
| Function | Purpose | Notes |
|----------+---------+-------|
| =lrg_widget_resolve_look (self, part, &look)= | Look of =part= now: defaults + rules for resting, hovered and pressed variants, blended by the springs | Call from =paint()= |
| =lrg_widget_resolve_look_for_state (self, part, state, &look)= | Defaults + rules for one exact state, no blending | Previews, tests |
| =lrg_widget_paint_box (self, list, &look, x, y, w, h)= | Draw the standard surface: glow, shadow, fill / gradient, inner border, at the look's opacity | |
| =lrg_widget_paint (self, list)= | Record the widget: opacity push, appear group, =paint()= (or the legacy =draw()= through a callback), focus ring when =FOCUS_VISIBLE= | Does nothing when invisible |
| =lrg_widget_draw (self)= | Legacy: draw immediately | Renders =paint()= output with a private renderer for paint-only widgets |
| =lrg_widget_handle_event (self, event)= | Call =handle_event()= if visible and enabled | |

=lrg_widget_paint()= in detail:

1. returns if invisible; starts a pending appear animation;
2. pushes the widget's opacity (when below 1);
3. marks a paint group;
4. records the widget: =paint()=, or for a class that overrides =draw()= but not =paint()=, a paint callback that calls =draw()=;
5. when =FOCUS_VISIBLE=: resolves the =FOCUS= look and draws a ring of its border width, 2 units outside the rectangle, with its radii;
6. while the appear animation runs: transforms the group (scale around the centre, offset, fade);
7. pops the opacity.

** Example: a minimal paint-based widget
:PROPERTIES:
:CUSTOM_ID: widget-example
:END:
A coloured "chip" with a label; the full walkthrough (keyboard, properties, YAML styling) is in [[file:../../guides/extending-ui.org][extending the UI]].

#+begin_src c
struct _GameChip
{
	LrgWidget  parent_instance;
	gchar     *text;
};

G_DEFINE_FINAL_TYPE (GameChip, game_chip, LRG_TYPE_WIDGET)

static void
game_chip_style_defaults (LrgWidget      *widget,
                          LrgUIPart       part,
                          LrgWidgetState  state,
                          LrgPartLook    *look)
{
	LrgTheme *theme = lrg_widget_get_theme (widget);

	if (part == LRG_UI_PART_BOX)
	{
		lrg_part_look_set_background (look, lrg_theme_get_color (theme,
			(state & LRG_WIDGET_STATE_HOVERED) ? LRG_THEME_COLOR_CONTROL_HOVER
			                                   : LRG_THEME_COLOR_CONTROL));
		lrg_part_look_set_radius (look, G_MAXFLOAT);          /* pill */
	}
	else if (part == LRG_UI_PART_LABEL)
		lrg_part_look_set_foreground (look, lrg_theme_get_color (theme, LRG_THEME_COLOR_TEXT));
	else
		LRG_WIDGET_CLASS (game_chip_parent_class)->style_defaults (widget, part, state, look);
}

static void
game_chip_measure (LrgWidget *widget,
                   gfloat    *w,
                   gfloat    *h)
{
	GameChip *self = GAME_CHIP (widget);
	LrgTheme *theme = lrg_widget_get_theme (widget);
	gfloat    size = lrg_theme_get_metric (theme, LRG_THEME_METRIC_FONT_SIZE_SMALL);
	gfloat    tw;
	gfloat    th;

	lrg_paint_measure_text (lrg_theme_get_ui_font (theme), self->text, size, 0.0f, &tw, &th);
	*w = tw + 2.0f * th;
	*h = th * 1.8f;
}

static void
game_chip_paint (LrgWidget    *widget,
                 LrgPaintList *list)
{
	GameChip   *self = GAME_CHIP (widget);
	LrgTheme   *theme = lrg_widget_get_theme (widget);
	gfloat      size = lrg_theme_get_metric (theme, LRG_THEME_METRIC_FONT_SIZE_SMALL);
	gfloat      x = lrg_widget_get_world_x (widget);
	gfloat      y = lrg_widget_get_world_y (widget);
	gfloat      w = lrg_widget_get_width (widget);
	gfloat      h = lrg_widget_get_height (widget);
	gfloat      tw;
	gfloat      th;
	LrgPartLook box;
	LrgPartLook label;

	lrg_widget_resolve_look (widget, LRG_UI_PART_BOX, &box);
	lrg_widget_resolve_look (widget, LRG_UI_PART_LABEL, &label);
	lrg_widget_paint_box (widget, list, &box, x, y, w, h);

	lrg_paint_measure_text (lrg_theme_get_ui_font (theme), self->text, size, 0.0f, &tw, &th);
	lrg_paint_list_text (list, lrg_theme_get_ui_font (theme), self->text,
	                     x + (w - tw) * 0.5f, y + (h - th) * 0.5f, size, 0.0f, &label.foreground);
}

static void
game_chip_class_init (GameChipClass *klass)
{
	LrgWidgetClass *widget_class = LRG_WIDGET_CLASS (klass);

	/* finalize (freeing text) omitted for brevity */
	widget_class->paint = game_chip_paint;
	widget_class->style_defaults = game_chip_style_defaults;
	widget_class->measure = game_chip_measure;
}

static void
game_chip_init (GameChip *self)
{
	self->text = g_strdup ("");
}
#+end_src

** Notes
:PROPERTIES:
:CUSTOM_ID: widget-notes
:END:
- Paint at world coordinates; =x= / =y= are parent-relative.
- Old widgets that implement only =draw()= keep working: they are recorded as paint callbacks (each one ends a paint batch) and do not get appear transforms. Port them to =paint()= (see [[file:../../guides/ui-migration.org][the migration guide]]).
- Hovering no longer sends =FOCUS_IN= / =FOCUS_OUT=; use =POINTER_ENTER= / =POINTER_LEAVE= or the =HOVERED= state.
- Call =lrg_widget_queue_layout()= whenever something that changes the preferred size changes.
