** Example: Character Controller
:PROPERTIES:
:CUSTOM_ID: example-character-controller
:END:
#+begin_src C
g_autoptr(LrgAnimationStateMachine) fsm = lrg_animation_state_machine_new ();
lrg_animation_state_machine_set_skeleton (fsm, skeleton);

/* Create states */
LrgAnimationState *idle = lrg_animation_state_new ("idle");
lrg_animation_state_set_clip (idle, idle_clip);

LrgAnimationState *walk = lrg_animation_state_new ("walk");
lrg_animation_state_set_clip (walk, walk_clip);

LrgAnimationState *run = lrg_animation_state_new ("run");
lrg_animation_state_set_clip (run, run_clip);

LrgAnimationState *jump = lrg_animation_state_new ("jump");
lrg_animation_state_set_clip (jump, jump_clip);
lrg_animation_state_set_loop (jump, FALSE);

/* Add states */
lrg_animation_state_machine_add_state (fsm, idle);
lrg_animation_state_machine_add_state (fsm, walk);
lrg_animation_state_machine_add_state (fsm, run);
lrg_animation_state_machine_add_state (fsm, jump);

/* Create transitions */

/* idle -> walk when speed > 0.1 */
LrgAnimationTransition *t1 = lrg_animation_transition_new ("idle", "walk");
lrg_animation_transition_add_condition (t1, "speed", LRG_CONDITION_GREATER, 0.1f);
lrg_animation_transition_set_duration (t1, 0.25f);

/* walk -> idle when speed < 0.1 */
LrgAnimationTransition *t2 = lrg_animation_transition_new ("walk", "idle");
lrg_animation_transition_add_condition (t2, "speed", LRG_CONDITION_LESS, 0.1f);
lrg_animation_transition_set_duration (t2, 0.25f);

/* walk -> run when speed > 0.6 */
LrgAnimationTransition *t3 = lrg_animation_transition_new ("walk", "run");
lrg_animation_transition_add_condition (t3, "speed", LRG_CONDITION_GREATER, 0.6f);
lrg_animation_transition_set_duration (t3, 0.2f);

/* any -> jump on trigger */
LrgAnimationTransition *t4 = lrg_animation_transition_new_any ("jump");
lrg_animation_transition_add_trigger (t4, "jump");
lrg_animation_transition_set_duration (t4, 0.1f);

/* jump -> idle at exit */
LrgAnimationTransition *t5 = lrg_animation_transition_new ("jump", "idle");
lrg_animation_transition_set_has_exit_time (t5, TRUE);
lrg_animation_transition_set_duration (t5, 0.2f);

/* Add transitions */
lrg_animation_state_machine_add_transition (fsm, t1);
lrg_animation_state_machine_add_transition (fsm, t2);
lrg_animation_state_machine_add_transition (fsm, t3);
lrg_animation_state_machine_add_transition (fsm, t4);
lrg_animation_state_machine_add_transition (fsm, t5);

/* Set default and start */
lrg_animation_state_machine_set_default_state (fsm, "idle");
lrg_animation_state_machine_start (fsm);
#+end_src

** Example: Game Loop
:PROPERTIES:
:CUSTOM_ID: example-game-loop
:END:
#+begin_src C
void
update_character (gfloat delta_time)
{
    /* Update parameters from game state */
    lrg_animation_state_machine_set_float (fsm, "speed", character_speed);
    lrg_animation_state_machine_set_bool (fsm, "grounded", is_grounded);

    /* Check for jump input */
    if (jump_pressed && is_grounded)
    {
        lrg_animation_state_machine_set_trigger (fsm, "jump");
    }

    /* Update state machine */
    lrg_animation_state_machine_update (fsm, delta_time);

    /* Apply to skeleton */
    lrg_skeleton_calculate_world_poses (skeleton);
}
#+end_src

** 0.9.0: typed parameters, any-state, interruption, sub-machines
:PROPERTIES:
:CUSTOM_ID: v090
:END:
The machine is the runtime of an [[file:anim-graph.org][LrgAnimGraph]] layer; everything a =.ranim= file says about a machine has an API here. The update order, exit time semantics, trigger consumption and interruption rules are described in [[file:anim-graph.org::#transitions][the graph page]].

*** Parameters
:PROPERTIES:
:CUSTOM_ID: v090-parameters
:END:
| function | |
|-+-|
| =lrg_animation_state_machine_add_parameter (self, name, type, default)= | declare a typed parameter (=LrgAnimParameterType=: =FLOAT=, =INT=, =BOOL=, =TRIGGER=; =lrg_anim_parameter_type_get_type()=) and set it to its default |
| =lrg_animation_state_machine_remove_parameter()= / =lrg_animation_state_machine_has_parameter()= | |
| =lrg_animation_state_machine_get_parameter_type()= / =lrg_animation_state_machine_get_parameter_names()= | declaration order |
| =lrg_animation_state_machine_reset_parameters()= | back to the defaults |
| =lrg_animation_state_machine_share_parameters (self, owner)= | use another machine's table (graph layers and sub-machines share one) |
| =lrg_animation_state_machine_set_int()= / =lrg_animation_state_machine_get_int()= | ints; =get_float= also reads ints |

Only declared triggers are consumed by the transitions they fire; undeclared parameters still work as before.

*** States, transitions, evaluation
:PROPERTIES:
:CUSTOM_ID: v090-runtime
:END:
| function | |
|-+-|
| =lrg_animation_state_machine_get_states_ordered()= | states in the order they were added |
| =lrg_animation_state_machine_remove_transition()= | |
| =lrg_animation_state_machine_get_next_state()= / =lrg_animation_state_machine_get_transition_progress()= / =lrg_animation_state_machine_get_active_transition()= | the cross-fade in progress |
| =lrg_animation_state_machine_sample_pose (self, skeleton, poses, n)= | the whole skeleton, without touching it |
| =lrg_animation_state_machine_get_root_motion (self, bone, translation, yaw)= | the last update's root motion, blended by the transition weight |

A transition whose source is =LRG_ANIMATION_TRANSITION_ANY_STATE= (="*"=) is an any-state transition. New transition API: =lrg_animation_transition_set_source()=, =lrg_animation_transition_set_target()=, =lrg_animation_transition_is_any_state()=, =lrg_animation_transition_get_offset()= / =lrg_animation_transition_set_offset()=, =lrg_animation_transition_get_interruption()= / =lrg_animation_transition_set_interruption()= (=LrgTransitionInterruption=, =lrg_transition_interruption_get_type()=: =NONE=, =SOURCE=, =DESTINATION=, =SOURCE_THEN_DESTINATION=, =DESTINATION_THEN_SOURCE=), =lrg_animation_transition_get_can_transition_to_self()= / =lrg_animation_transition_set_can_transition_to_self()= and =lrg_animation_transition_evaluate_range (self, parameters, previous_normalized_time, normalized_time)= (crossing semantics for the exit time; =lrg_animation_transition_evaluate()= keeps the old "at or past" test). The exit time may now be above 1 (whole loops), priorities are honoured (higher first) and source/target are writable properties.

New state API: =lrg_animation_state_get_blend_tree()= / =lrg_animation_state_set_blend_tree()=, =lrg_animation_state_get_sub_machine()= / =lrg_animation_state_set_sub_machine()=, =lrg_animation_state_get_clip_name()= / =lrg_animation_state_set_clip_name()= (the graph's clip key), =lrg_animation_state_get_speed_parameter()= / =lrg_animation_state_set_speed_parameter()=, =lrg_animation_state_get_previous_time()=, =lrg_animation_state_get_duration()=, =lrg_animation_state_sample_pose (self, skeleton, poses, n)= and =lrg_animation_state_get_root_motion (self, bone, translation, yaw)=. The per-bone =lrg_animation_state_sample()= now returns the identity for a bone without a track (it used to sample track 0).

*** New signals
:PROPERTIES:
:CUSTOM_ID: v090-signals
:END:
- =event (state_name, LrgAnimationEvent *event)=, detailed by the event name: every clip event an update plays past, of the current state, of the state blending in and (forwarded) of nested machines; blend trees fire their dominant clip's.
- =transition-started (source, target)=.

=state-entered= is still emitted when a state becomes current (the end of the cross-fade).

*** Standalone use
:PROPERTIES:
:CUSTOM_ID: v090-standalone
:END:
With a =skeleton= set, every update samples the machine and writes the pose into the skeleton as before (through =lrg_anim_pose_apply()=, bind pose for untracked bones). Machines inside a graph have no skeleton; the graph samples them.
